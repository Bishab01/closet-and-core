Create a file named `connectDB.php` in this folder and replace the placeholder values with your own MySQL database credentials.

<?php
// Connecting to the database
$host = "127.0.0.1";
$user = "root";
$password = "your_root_password";
$dbname = "your_database_name";
$port = 3307; // Use the port your MySQL server is running on (default is 3306)

$conn = new mysqli($host, $user, $password, $dbname, $port);

if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}
?>
